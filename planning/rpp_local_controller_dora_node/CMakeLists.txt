cmake_minimum_required(VERSION 3.16)
project(rpp_local_controller_dora_node LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(DORA_INCLUDE_DIR "")
set(DORA_LIB_PATH "")
foreach(_base "$ENV{HOME}/dora" "$ENV{HOME}/dora-main" "$ENV{HOME}/Public/dora")
  if(EXISTS "${_base}/apis/c/node/node_api.h" AND NOT DORA_INCLUDE_DIR)
    set(DORA_INCLUDE_DIR "${_base}/apis/c/node")
  endif()
endforeach()
foreach(_lib "$ENV{HOME}/dora/target/release/libdora_node_api_c.a" "$ENV{HOME}/dora-main/target/release/libdora_node_api_c.a" "$ENV{HOME}/Public/dora/target/release/libdora_node_api_c.a")
  if(EXISTS "${_lib}" AND NOT DORA_LIB_PATH)
    set(DORA_LIB_PATH "${_lib}")
  endif()
endforeach()
if(DORA_INCLUDE_DIR AND DORA_LIB_PATH)
 add_executable(rpp_local_controller_dora_node src/rpp_local_controller_dora_node.cpp
   src/path_handler.cpp src/path_json.cpp src/rpp_controller.cpp src/costmap.cpp src/rotation_shim.cpp)
 target_include_directories(rpp_local_controller_dora_node PRIVATE ${DORA_INCLUDE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/include)
 target_link_libraries(rpp_local_controller_dora_node PRIVATE ${DORA_LIB_PATH})
endif()
enable_testing()
add_executable(path_handler_test tests/path_handler_test.cpp src/path_handler.cpp)
target_include_directories(path_handler_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_test(NAME path_handler COMMAND path_handler_test)
add_executable(rpp_controller_test tests/rpp_controller_test.cpp src/rpp_controller.cpp src/costmap.cpp)
target_include_directories(rpp_controller_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_test(NAME rpp_controller COMMAND rpp_controller_test)
add_executable(path_json_contract_test tests/path_json_contract_test.cpp src/path_json.cpp)
target_include_directories(path_json_contract_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_test(NAME path_json_contract COMMAND path_json_contract_test)
